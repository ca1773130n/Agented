**Step 5: Sanity test** — write `backend/tests/test_schema_split.py`:

```python
"""v0.7.3: confirm schema split produces identical table set."""
from app.database import get_connection
from app.db.schema import create_fresh_schema


def test_create_fresh_schema_idempotent(isolated_db):
    """Running twice produces no errors."""
    with get_connection() as conn:
        create_fresh_schema(conn)


def test_all_tables_present(isolated_db):
    """A baseline of well-known tables must exist after create_fresh_schema."""
    with get_connection() as conn:
        cursor = conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table' "
            "AND name NOT LIKE 'sqlite_%'"
        )
        tables = {row["name"] for row in cursor.fetchall()}
    expected = {
        "triggers", "execution_logs", "agents", "agent_conversations",
        "user_skills", "teams", "team_members", "products", "projects",
        "plugins", "marketplaces", "users", "sessions",
    }
    missing = expected - tables
    assert not missing, f"Missing expected tables: {missing}"
```

"""v0.7.3: ensure split migrations preserve version ordering."""
from app.db.migrations import VERSIONED_MIGRATIONS


def test_versions_strictly_increasing():
    versions = [v for v, _, _ in VERSIONED_MIGRATIONS]
    assert versions == sorted(set(versions)), \
        f"Migration versions must be strictly increasing & unique; got {versions[:10]}…"


def test_no_duplicate_names():
    names = [n for _, n, _ in VERSIONED_MIGRATIONS]
    assert len(names) == len(set(names)), "Migration names must be unique"

    from app.db.migrations import run_migrations
    assert callable(run_migrations)
